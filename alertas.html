"use strict"; // CENTRAL DE ALERTAS - ESTOQUE SIMPLES const tabela = document.getElementById("tabelaAlertas"); const pesquisa = document.getElementById("pesquisa"); const filtro = document.getElementById("filtro"); const ordenacao = document.getElementById("ordenacao"); const seguranca = document.getElementById("seguranca"); const totalCriticos = document.getElementById("totalCriticos"); const totalVencidos = document.getElementById("totalVencidos"); const totalProximos = document.getElementById("totalProximos"); const totalRegulares = document.getElementById("totalRegulares"); let resultados = []; // LER PRODUTOS EXISTENTES function obterProdutos() { try { const dados = JSON.parse( localStorage.getItem("produtos") || "[]" ); return Array.isArray(dados) ? dados : []; } catch (erro) { console.error("Erro ao carregar produtos:", erro); return []; } } // CALCULAR ESTOQUE ATUAL function estoqueAtual(produto) { const inicial = Number( produto.estoqueInicial ?? produto.quantidade ?? 0 ); const entradas = Number(produto.totalEntradas || 0); const saidas = Number(produto.totalSaidas || 0); return inicial + entradas - saidas; } // CONVERTER DATA SEM PROBLEMAS DE FUSO HORÁRIO function converterData(valor) { if (!valor) { return null; } const texto = String(valor).trim(); let ano, mes, dia; if (/^\d{4}-\d{2}-\d{2}/.test(texto)) { const partes = texto.substring(0, 10).split("-"); ano = Number(partes[0]); mes = Number(partes[1]); dia = Number(partes[2]); } else if (/^\d{2}\/\d{2}\/\d{4}$/.test(texto)) { const partes = texto.split("/"); dia = Number(partes[0]); mes = Number(partes[1]); ano = Number(partes[2]); } else { return null; } const data = new Date(ano, mes - 1, dia); if ( data.getFullYear() !== ano || data.getMonth() !== mes - 1 || data.getDate() !== dia ) { return null; } data.setHours(0, 0, 0, 0); return data; } function hojeSemHorario() { const hoje = new Date(); hoje.setHours(0, 0, 0, 0); return hoje; } function diferencaDias(data) { const hoje = hojeSemHorario(); // Utilizar UTC para comparar apenas datas civis. const atualUTC = Date.UTC( hoje.getFullYear(), hoje.getMonth(), hoje.getDate() ); const validadeUTC = Date.UTC( data.getFullYear(), data.getMonth(), data.getDate() ); return Math.round( (validadeUTC - atualUTC) / 86400000 ); } function formatarData(data) { if (!data) { return "Não informada"; } return String(data.getDate()).padStart(2, "0") + "/" + String(data.getMonth() + 1).padStart(2, "0") + "/" + data.getFullYear(); } function formatarNumero(valor) { return Number(valor).toLocaleString("pt-BR", { maximumFractionDigits: 3 }); } // CLASSIFICAR CADA PRODUTO function analisarProduto(produto) { const quantidade = estoqueAtual(produto); const minimo = Number(produto.estoqueMinimo || 0); const validade = converterData(produto.validade); const dias = validade === null ? null : diferencaDias(validade); const critico = quantidade <= minimo; // Produtos sem unidades disponíveis não // são contados como mercadorias vencidas. const possuiEstoque = quantidade > 0; const vencido = possuiEstoque && dias !== null && dias < 0; const proximo = possuiEstoque && dias !== null && dias >= 0 && dias <= 7; const regular = !critico && !vencido && !proximo; const adicional = Math.max( 0, Math.floor(Number(seguranca.value) || 0) ); const reposicao = critico ? Math.max( 0, Math.ceil(minimo + adicional - quantidade) ) : 0; let prioridade = "Baixa"; let ordem = 3; if (vencido) { prioridade = "Alta"; ordem = 0; } else if (critico) { prioridade = "Alta"; ordem = 1; } else if (proximo) { prioridade = "Média"; ordem = 2; } const situacoes = []; if (critico) { situacoes.push({ nome: "Estoque crítico", classe: "critico" }); } if (vencido) { situacoes.push({ nome: "Vencido", classe: "vencido" }); } if (proximo) { situacoes.push({ nome: "Próximo do vencimento", classe: "proximo" }); } if (regular) { situacoes.push({ nome: "Regular", classe: "regular" }); } return { nome: String(produto.nome || "Sem nome"), codigo: String(produto.codigo || ""), quantidade: quantidade, minimo: minimo, validade: validade, critico: critico, vencido: vencido, proximo: proximo, regular: regular, prioridade: prioridade, ordem: ordem, reposicao: reposicao, situacoes: situacoes }; } // ANALISAR TODOS OS PRODUTOS ATIVOS function atualizarAnalise() { const produtos = obterProdutos().filter( function(produto) { return produto.ativo !== false; } ); resultados = produtos.map(analisarProduto); totalCriticos.textContent = resultados.filter( item => item.critico ).length; totalVencidos.textContent = resultados.filter( item => item.vencido ).length; totalProximos.textContent = resultados.filter( item => item.proximo ).length; totalRegulares.textContent = resultados.filter( item => item.regular ).length; atualizarTabela(); } // FILTRAR E ORDENAR function obterResultadosFiltrados() { const termo = pesquisa.value.trim().toLowerCase(); const tipo = filtro.value; const ordemEscolhida = ordenacao.value; const lista = resultados.filter(function(item) { const texto = ( item.nome + " " + item.codigo ).toLowerCase(); if (!texto.includes(termo)) { return false; } if (tipo === "critico") { return item.critico; } if (tipo === "vencido") { return item.vencido; } if (tipo === "proximo") { return item.proximo; } if (tipo === "regular") { return item.regular; } return true; }); lista.sort(function(a, b) { if (ordemEscolhida === "nome") { return a.nome.localeCompare(b.nome, "pt-BR"); } if (ordemEscolhida === "validade") { const dataA = a.validade ? a.validade.getTime() : Infinity; const dataB = b.validade ? b.validade.getTime() : Infinity; return dataA - dataB; } return a.ordem - b.ordem || a.nome.localeCompare(b.nome, "pt-BR"); }); return lista; } // ADICIONAR CÉLULA DE FORMA SEGURA function adicionarCelula(linha, texto) { const celula = document.createElement("td"); celula.textContent = texto; linha.appendChild(celula); return celula; } // DESENHAR TABELA function atualizarTabela() { tabela.innerHTML = ""; const lista = obterResultadosFiltrados(); if (lista.length === 0) { const linha = document.createElement("tr"); const celula = document.createElement("td"); celula.colSpan = 8; celula.textContent = "Nenhum produto encontrado."; linha.appendChild(celula); tabela.appendChild(linha); return; } lista.forEach(function(item) { const linha = document.createElement("tr"); adicionarCelula(linha, item.nome); adicionarCelula(linha, item.codigo); adicionarCelula( linha, formatarNumero(item.quantidade) ); adicionarCelula( linha, formatarNumero(item.minimo) ); adicionarCelula( linha, formatarData(item.validade) ); const situacao = document.createElement("td"); item.situacoes.forEach(function(alerta) { const etiqueta = document.createElement("span"); etiqueta.className = "etiqueta " + alerta.classe; etiqueta.textContent = alerta.nome; etiqueta.style.marginRight = "5px"; etiqueta.style.marginBottom = "5px"; situacao.appendChild(etiqueta); }); linha.appendChild(situacao); const prioridade = adicionarCelula( linha, item.prioridade ); prioridade.className = "prioridade " + ( item.prioridade === "Alta" ? "alta" : item.prioridade === "Média" ? "media" : "baixa" ); adicionarCelula( linha, item.reposicao > 0 ? formatarNumero(item.reposicao) + " un." : "Não necessária" ); tabela.appendChild(linha); }); } // EVENTOS pesquisa.addEventListener("input", atualizarTabela); filtro.addEventListener("change", atualizarTabela); ordenacao.addEventListener("change", atualizarTabela); document.getElementById("atualizar") .addEventListener("click", function() { atualizarAnalise(); }); seguranca.addEventListener("change", atualizarAnalise); // EXPORTAR CSV function protegerCSV(valor) { const texto = String(valor ?? ""); if (/^\s*[=+\-@\t\r]/.test(texto)) { return "'" + texto; } return texto; } function celulaCSV(valor) { return '"' + protegerCSV(valor).replace(/"/g, '""') + '"'; } document.getElementById("exportar") .addEventListener("click", function() { const cabecalho = [ "Produto", "Código", "Estoque atual", "Estoque mínimo", "Validade", "Situação", "Prioridade", "Reposição sugerida" ]; const linhas = [cabecalho]; obterResultadosFiltrados().forEach(function(item) { linhas.push([ item.nome, item.codigo, item.quantidade, item.minimo, formatarData(item.validade), item.situacoes.map( alerta => alerta.nome ).join(" / "), item.prioridade, item.reposicao ]); }); const csv = "\uFEFF" + linhas.map( linha => linha.map(celulaCSV).join(";") ).join("\r\n"); const arquivo = new Blob([csv], { type: "text/csv;charset=utf-8;" }); const url = URL.createObjectURL(arquivo); const link = document.createElement("a"); link.href = url; link.download = "relatorio_alertas.csv"; document.body.appendChild(link); link.click(); link.remove(); URL.revokeObjectURL(url); }); // ATUALIZAR AO RETORNAR PARA A PÁGINA window.addEventListener("pageshow", atualizarAnalise); // INICIAR atualizarAnalise();
